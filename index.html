import java.util.Scanner;
import java.util.Arrays;

public class ReliefGoodAnalyzer {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        int[] packages = new int[12];
        int total = 0;
        int highest = 0;

        System.out.println("RELIEF GOODS DISTRIBUTION ANALYZER");

        // Input distribution counts
        for (int i = 0; i < 12; i++) {
            System.out.print("Enter packages for Community " + (i + 1) + ": ");
            packages[i] = input.nextInt();

            while (packages[i] < 0) {
                System.out.print("Invalid! Enter a nonnegative number: ");
                packages[i] = input.nextInt();
            }

            total += packages[i];

            if (packages[i] > highest) {
                highest = packages[i];
            }
        }

        // Calculate average
        double average = total / 12.0;

        // Find communities with the most packages
        System.out.println("\nCommunity/ies receiving the most packages:");
        for (int i = 0; i < 12; i++) {
            if (packages[i] == highest) {
                System.out.println("Community " + (i + 1));
            }
        }

        // Sort in descending order
        Arrays.sort(packages);

        System.out.println("\nDistribution Counts (Descending Order):");
        for (int i = 11; i >= 0; i--) {
            System.out.print(packages[i] + " ");
        }

        // Display results
        System.out.println("\n\nTotal Packages: " + total);
        System.out.println("Average Packages: " + average);

        input.close();
    }
}
